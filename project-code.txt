import Fastify from "fastify";
import { ZodError } from "zod";
import { accountRoutes } from "./modules/accounts/accounts.routes";
import { currentAccountRoutes } from "./modules/accounts/current-account.routes";
import { consumeRoutes } from "./modules/consume/consume.routes";
import { planRoutes } from "./modules/plans/plans.routes";
import { userRoutes } from "./modules/users/users.routes";

export function buildApp() {
  const app = Fastify({ logger: process.env.NODE_ENV !== "test" });

  app.setErrorHandler((err: unknown, req, reply) => {
    if (err instanceof ZodError) {
      return reply.code(400).send({ error: "validation_error", details: err.issues });
    }
    const e = err as { statusCode?: number; message?: string };
    if (typeof e.statusCode === "number" && e.statusCode < 500) {
      return reply.code(e.statusCode).send({ error: e.message ?? "bad_request" });
    }
    req.log.error(err);
    return reply.code(500).send({ error: "internal_error" });
  });

  app.get("/health", async () => ({ ok: true }));
  app.register(accountRoutes);
  app.register(currentAccountRoutes);
  app.register(planRoutes);
  app.register(userRoutes);
  app.register(consumeRoutes);

  return app;
}

import { createHash, randomBytes } from "node:crypto";
import type { FastifyReply, FastifyRequest } from "fastify";
import { and, eq, isNull } from "drizzle-orm";
import { db } from "./db/client";
import { apiKeys } from "./db/schema";

declare module "fastify" {
  interface FastifyRequest {
    accountId: string;
  }
}

export const generateApiKey = () => "qk_" + randomBytes(32).toString("hex");
export const hashKey = (key: string) => createHash("sha256").update(key).digest("hex");

export async function requireApiKey(req: FastifyRequest, reply: FastifyReply) {
  const key = req.headers["x-api-key"];
  if (typeof key !== "string" || key.length === 0) {
    return reply.code(401).send({ error: "missing_api_key" });
  }
  const [row] = await db
    .select({ accountId: apiKeys.accountId })
    .from(apiKeys)
    .where(and(eq(apiKeys.keyHash, hashKey(key)), isNull(apiKeys.revokedAt)))
    .limit(1);
  if (!row) return reply.code(401).send({ error: "invalid_api_key" });
  req.accountId = row.accountId;
}

import type { FastifyInstance } from "fastify";
import { and, eq, isNull } from "drizzle-orm";
import { z } from "zod";
import { db } from "../../db/client";
import { accounts, apiKeys, plans } from "../../db/schema";
import { PLAN_IDS } from "../../db/seed";
import { generateApiKey, hashKey } from "../../auth";

const body = z.object({
  name: z.string().trim().min(1).max(100),
  planId: z.guid().optional(),
});

export async function accountRoutes(app: FastifyInstance) {
  app.post("/accounts", async (req, reply) => {
    const { name, planId = PLAN_IDS.accountStarter } = body.parse(req.body);

    const [plan] = await db
      .select({ id: plans.id })
      .from(plans)
      .where(and(eq(plans.id, planId), eq(plans.scope, "account"), isNull(plans.accountId)))
      .limit(1);
    if (!plan) return reply.code(400).send({ error: "invalid_plan" });

    const apiKey = generateApiKey();
    const account = await db.transaction(async (tx) => {
      const [acc] = await tx.insert(accounts).values({ name, planId }).returning();
      await tx.insert(apiKeys).values({
        accountId: acc.id,
        keyHash: hashKey(apiKey),
        keyPrefix: apiKey.slice(0, 11),
      });
      return acc;
    });

    return reply.code(201).send({ account, apiKey });
  });
}

import type { FastifyInstance } from "fastify";

export async function currentAccountRoutes(app: FastifyInstance) {
  app.addHook("preHandler", requireApiKey);

  const load = async (accountId: string) => {
    const [row] = await db
      .select({ account: accounts, plan: plans })
      .from(accounts)
      .innerJoin(plans, eq(accounts.planId, plans.id))
      .where(eq(accounts.id, accountId))
      .limit(1);
    return row;
  };

  app.get("/account", async (req) => load(req.accountId));

  });
}

import { sql } from "drizzle-orm";
import { db } from "../../db/client";
import { evaluate, remaining, type Level, type Rejection, type RuleName, type Snapshot } from "./rules";

export type ConsumeInput = {
  accountId: string;
  userExternalId: string;
  units: number;
  requestId: string;
};

export type Balance = Snapshot & { remaining: number };

export type ConsumeResult =
  | { kind: "user_not_found" }
  | { kind: "request_id_reused" }
  | {
      kind: "decided";
      replayed: boolean;
      accepted: boolean;
      rejection: Rejection | null;
      units: number;
      month: string;
      account: Balance;
      user: Balance;
    };

type UsedRow = { used: string };
type PrevRow = {
  user_id: string;
  units: string;
  accepted: boolean;
  rejected_level: Level | null;
  rejected_rule: RuleName | null;
};
type LimitsRow = { a_month: string; a_max: string; u_month: string; u_max: string };

const balance = (s: Snapshot): Balance => ({ ...s, remaining: remaining(s) });

export const monthStart = (d: Date) =>
  `${d.getUTCFullYear()}-${String(d.getUTCMonth() + 1).padStart(2, "0")}-01`;

export async function consume(input: ConsumeInput, now: Date = new Date()): Promise<ConsumeResult> {
  const { accountId, userExternalId, units, requestId } = input;
  const month = monthStart(now);

  return db.transaction(async (tx): Promise<ConsumeResult> => {
    const found = await tx.execute<{ id: string }>(
      sql`SELECT id FROM users WHERE account_id = ${accountId} AND external_id = ${userExternalId}`,
    );
    const userId = found.rows[0]?.id;
    if (!userId) return { kind: "user_not_found" };

    // ترتيب الأقفال ثابت دايماً: صف الحساب ثم صف المستخدم (بيمنع deadlock).
    // الـ upsert "الفارغ" بينشئ الصف إذا مش موجود، وبيقفله، وبيرجّع القيمة، بطلب واحد.
    const acc = await tx.execute<UsedRow>(sql`
      INSERT INTO account_usage (account_id, month, used)
      VALUES (${accountId}, ${month}::date, 0)
      ON CONFLICT (account_id, month) DO UPDATE SET used = account_usage.used
      RETURNING used`);
    const usr = await tx.execute<UsedRow>(sql`
      INSERT INTO user_usage (user_id, month, used)
      VALUES (${userId}, ${month}::date, 0)
      ON CONFLICT (user_id, month) DO UPDATE SET used = user_usage.used
      RETURNING used`);

    const prev = await tx.execute<PrevRow>(sql`
      SELECT user_id, units, accepted, rejected_level, rejected_rule
      FROM request_log
      WHERE account_id = ${accountId} AND request_id = ${requestId}`);

    const lim = (
      await tx.execute<LimitsRow>(sql`
        SELECT ap.monthly_allowance AS a_month, ap.max_per_request AS a_max,
               up.monthly_allowance AS u_month, up.max_per_request AS u_max
        FROM accounts a
        JOIN plans ap ON ap.id = a.plan_id
        JOIN users u  ON u.id = ${userId}
        JOIN plans up ON up.id = u.plan_id
        WHERE a.id = ${accountId}`)
    ).rows[0];

    const account: Snapshot = {
      monthlyAllowance: Number(lim.a_month),
      maxPerRequest: Number(lim.a_max),
      used: Number(acc.rows[0].used),
    };
    const user: Snapshot = {
      monthlyAllowance: Number(lim.u_month),
      maxPerRequest: Number(lim.u_max),
      used: Number(usr.rows[0].used),
    };

    const old = prev.rows[0];
    if (old) {
      if (old.user_id !== userId || Number(old.units) !== units) return { kind: "request_id_reused" };
      return {
        kind: "decided",
        replayed: true,
        accepted: old.accepted,
        rejection:
          old.rejected_level && old.rejected_rule
            ? { level: old.rejected_level, rule: old.rejected_rule }
            : null,
        units,
        month,
        account: balance(account),
        user: balance(user),
      };
    }

    const rejection = evaluate(units, { account, user });

    if (!rejection) {
      await tx.execute(sql`
        UPDATE account_usage SET used = used + ${units}
        WHERE account_id = ${accountId} AND month = ${month}::date`);
      await tx.execute(sql`
        UPDATE user_usage SET used = used + ${units}
        WHERE user_id = ${userId} AND month = ${month}::date`);
      account.used += units;
      user.used += units;
    }

    await tx.execute(sql`
      INSERT INTO request_log
        (account_id, user_id, request_id, units, accepted, rejected_level, rejected_rule, month)
      VALUES
        (${accountId}, ${userId}, ${requestId}, ${units}, ${!rejection},
         ${rejection?.level ?? null}, ${rejection?.rule ?? null}, ${month}::date)`);

    return {
      kind: "decided",
      replayed: false,
      accepted: !rejection,
      rejection,
      units,
      month,
      account: balance(account),
      user: balance(user),
    };
  });
}


===== FILE: src/modules/consume/rules.ts =====

export type Level = "account" | "user";
export type RuleName = "monthly_allowance" | "max_per_request";

export type Snapshot = { monthlyAllowance: number; maxPerRequest: number; used: number };
export type Rejection = { level: Level; rule: RuleName };
type Context = { account: Snapshot; user: Snapshot };
type Check = (units: number, ctx: Context) => Rejection | null;

export const remaining = (s: Snapshot) => Math.max(0, s.monthlyAllowance - s.used);

const maxPerRequest =
  (level: Level): Check =>
  (units, ctx) =>
    units > ctx[level].maxPerRequest ? { level, rule: "max_per_request" } : null;

const monthlyAllowance =
  (level: Level): Check =>
  (units, ctx) =>
    units > remaining(ctx[level]) ? { level, rule: "monthly_allowance" } : null;

export const checks: Check[] = [
  maxPerRequest("user"),
  maxPerRequest("account"),
  monthlyAllowance("user"),
  monthlyAllowance("account"),
];

export function evaluate(units: number, ctx: Context): Rejection | null {
  for (const check of checks) {
    const rejection = check(units, ctx);
    if (rejection) return rejection;
  }
  return null;
}

export async function planRoutes(app: FastifyInstance) {

    const [plan] = await db

      .onConflictDoNothing()
      .returning();
    if (!plan) return reply.code(409).send({ error: "plan_name_taken" });

  });

  app.get("/plans", async (req) => {
    const rows = await db
      .select()
      .from(plans)
      .where(visible(req.accountId))
      .orderBy(asc(plans.scope), asc(plans.name));
    return { plans: rows };
  });

  app.get("/plans/:id", async (req, reply) => {
    const { id } = idParams.parse(req.params);
    const [plan] = await db
      .select()
      .from(plans)
      .where(and(eq(plans.id, id), visible(req.accountId)))
      .limit(1);
    if (!plan) return reply.code(404).send({ error: "plan_not_found" });
    return { plan };
  });

  app.patch("/plans/:id", async (req, reply) => {
    const { id } = idParams.parse(req.params);
    const changes = updateBody.parse(req.body);

    const [existing] = await db
      .select()
      .from(plans)
      .where(and(eq(plans.id, id), visible(req.accountId)))
      .limit(1);
    if (!existing) return reply.code(404).send({ error: "plan_not_found" });
    if (existing.accountId === null) {
      return reply.code(403).send({ error: "predefined_plan_read_only" });
    }

    try {
      const [plan] = await db
        .update(plans)
        .set({ ...changes, updatedAt: new Date() })
        .where(and(eq(plans.id, id), eq(plans.accountId, req.accountId)))
        .returning();
      return { plan };
    } catch (err) {
      if (isUniqueViolation(err)) return reply.code(409).send({ error: "plan_name_taken" });
      throw err;
    }
  });

import { and, eq, isNull, or } from "drizzle-orm";
import { db } from "../../db/client";
import { plans } from "../../db/schema";

export async function findAssignablePlan(
  accountId: string,
  planId: string,
  scope: "account" | "user",
) {
  const [plan] = await db
    .select()
    .from(plans)
    .where(
      and(
        eq(plans.id, planId),
        eq(plans.scope, scope),
        or(isNull(plans.accountId), eq(plans.accountId, accountId)),
      ),
    )
    .limit(1);
  return plan ?? null;
}

export function isUniqueViolation(err: unknown): boolean {
  const e = err as { code?: string; cause?: { code?: string } };
  return e?.code === "23505" || e?.cause?.code === "23505";
}

export async function userRoutes(app: FastifyInstance) {
  app.addHook("preHandler", requireApiKey);

  // الـ accountId دايماً من المفتاح، مش من الطلب
  const load = async (accountId: string, externalId: string) => {
    const [row] = await db
      .select({ user: users, plan: plans })
      .from(users)
      .innerJoin(plans, eq(users.planId, plans.id))
      .where(and(eq(users.accountId, accountId), eq(users.externalId, externalId)))
      .limit(1);
    return row;
  };

    const { externalId, planId = PLAN_IDS.userBasic } = createBody.parse(req.body);

    const plan = await findAssignablePlan(req.accountId, planId, "user");
    if (!plan) return reply.code(400).send({ error: "invalid_plan" });

    const [user] = await db
      .insert(users)
      .values({ accountId: req.accountId, externalId, planId })
      .onConflictDoNothing({ target: [users.accountId, users.externalId] })
      .returning();
    if (!user) return reply.code(409).send({ error: "user_already_exists" });

    return reply.code(201).send({ user, plan });
  });

  app.get("/users/:externalId", async (req, reply) => {
    const { externalId } = params.parse(req.params);
    const row = await load(req.accountId, externalId);
    if (!row) return reply.code(404).send({ error: "user_not_found" });
    return row;
  });

  app.put("/users/:externalId/plan", async (req, reply) => {
    const { externalId } = params.parse(req.params);
    const { planId } = planBody.parse(req.body);

    const existing = await load(req.accountId, externalId);
    if (!existing) return reply.code(404).send({ error: "user_not_found" });

    const plan = await findAssignablePlan(req.accountId, planId, "user");
    if (!plan) return reply.code(400).send({ error: "invalid_plan" });

    await db
      .update(users)
      .set({ planId })
      .where(and(eq(users.accountId, req.accountId), eq(users.externalId, externalId)));
    return load(req.accountId, externalId);
  });
}

describe("account isolation", () => {
  it("hides users and plans of other accounts", async () => {
    const keyA = await createAccount();
    const keyB = await createAccount();

    await call(keyA, "POST", "/users", { externalId: "alice" });
    const plan = (await call(keyA, "POST", "/plans", {

    expect((await call(keyB, "GET", "/users/alice")).statusCode).toBe(404);

  });

  it("allows the same external user id in different accounts", async () => {

    expect((await call(keyA, "POST", "/users", { externalId: "same" })).statusCode).toBe(201);

import { describe, expect, it } from "vitest";
import { evaluate } from "../src/modules/consume/rules";

const snap = (monthlyAllowance: number, maxPerRequest: number, used = 0) => ({
  monthlyAllowance, maxPerRequest, used,
});

describe("rules", () => {
  it("accepts when every rule passes, including exactly the remaining amount", () => {
    expect(evaluate(10, { account: snap(100, 50, 90), user: snap(100, 50, 90) })).toBeNull();
  });
  it("reports the user per-request rule first when several are violated", () => {
    const r = evaluate(25, { account: snap(5, 10), user: snap(1000, 20) });
    expect(r).toEqual({ level: "user", rule: "max_per_request" });
  });
  it("rejects at the account level when only the account is short", () => {
    const r = evaluate(10, { account: snap(5, 100), user: snap(1000, 20) });
    expect(r).toEqual({ level: "account", rule: "monthly_allowance" });
  });
  it("treats an allowance reduced below usage as zero remaining", () => {
    const r = evaluate(1, { account: snap(100, 10), user: snap(50, 10, 80) });
    expect(r).toEqual({ level: "user", rule: "monthly_allowance" });
  });
});
